async function startCamera() { if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { alert( "Camera API is not available. Open EcoSort AI through HTTPS in Chrome." ); return; } try { console.log("Requesting camera..."); const devices = await navigator.mediaDevices.enumerateDevices(); console.log( "Camera devices:", devices.filter( d => d.kind === "videoinput" ) ); stream = await navigator.mediaDevices.getUserMedia({ audio: false, video: { facingMode: { ideal: "environment" }, width: { ideal: 1280 }, height: { ideal: 720 } } }); console.log( "Camera stream started" ); video.srcObject = stream; await video.play(); running = true; startBtn.disabled = true; stopBtn.disabled = false; scanStatus.textContent = "ON"; systemStatus.textContent = "Scanning"; detectFrame(); } catch (error) { console.error( "CAMERA ERROR:", error ); let message = "Camera could not start.\n\n"; if ( error.name === "NotAllowedError" ) { message += "Camera permission was denied. Allow camera access in Chrome."; } else if ( error.name === "NotFoundError" ) { message += "No camera was found on this device."; } else if ( error.name === "NotReadableError" ) { message += "The camera is being used by another application."; } else if ( error.name === "SecurityError" ) { message += "The browser blocked camera access because of the page security settings."; } else { message += error.name + ": " + error.message; } alert(message); } }
